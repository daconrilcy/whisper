use sha2::{Digest, Sha256};
use std::env;
use std::fs::{self, File};
use std::io::{Read, Write};
use std::path::Path;
use std::io::BufWriter;
use rusty_mp3::{Mp3Encoder, Mp3EncoderConfig};

const DURATIONS: [u64; 3] = [14460, 6540, 6000];
fn hash(path: &Path) -> Result<String, Box<dyn std::error::Error>> {
    let mut file = File::open(path)?;
    let mut h = Sha256::new();
    let mut buf = [0_u8; 65536];
    loop { let n=file.read(&mut buf)?; if n==0 {break;} h.update(&buf[..n]); }
    Ok(h.finalize().iter().map(|b| format!("{b:02x}")).collect())
}
fn write_sync(path: &Path, bytes: &[u8]) -> Result<(), Box<dyn std::error::Error>> {
    let mut f=File::create(path)?; f.write_all(bytes)?; f.sync_all()?; Ok(())
}
fn hms(ms:u64)->String {format!("{:02}:{:02}:{:02},{:03}",ms/3600000,(ms/60000)%60,(ms/1000)%60,ms%1000)}
fn encode(source:&Path,index:usize,target:&Path)->Result<(),Box<dyn std::error::Error>> {
    let name=match index{1=>"fr_fr-000.wav",2=>"en_us-000.wav",3=>"fr_fr-001.wav",_=>return Err("bad index".into())};
    let mut reader=hound::WavReader::open(source.join(name))?;
    let spec=reader.spec();
    if spec.sample_rate!=16000 || spec.channels!=1{return Err("fixture format".into())}
    let pcm:Vec<i16>=match(spec.sample_format,spec.bits_per_sample){
        (hound::SampleFormat::Int,16)=>reader.samples::<i16>().collect::<Result<_,_>>()?,
        (hound::SampleFormat::Float,32)=>reader.samples::<f32>().map(|v|v.map(|x|(x.clamp(-1.0,1.0)*32767.0).round() as i16)).collect::<Result<_,_>>()?,
        _=>return Err("fixture sample type".into()),
    };
    let mut encoder=Mp3Encoder::new(Mp3EncoderConfig{bitrate_kbps:64,vbr_quality:None});
    let mut writer=BufWriter::new(File::create(target)?);
    for chunk in pcm.chunks(8000){
        encoder.push_pcm_s16(chunk,spec.channels,spec.sample_rate)?;
        while let Ok(packet)=encoder.next_packet(){writer.write_all(&packet)?;}
    }
    encoder.finish();
    while let Ok(packet)=encoder.next_packet(){writer.write_all(&packet)?;}
    writer.flush()?;writer.get_ref().sync_all()?;
    Ok(())
}
fn current(root:&Path)->Result<usize, Box<dyn std::error::Error>> {
    let p=root.join("current.txt");
    if !p.exists(){return Ok(0)}
    let v=fs::read_to_string(p)?;
    Ok(v.trim().parse()?)
}
fn scan(root:&Path)->Result<&'static str, Box<dyn std::error::Error>> {
    let index=match current(root){Ok(value)=>value,Err(_)=>return Ok("RECOVERABLE_BAD_CURRENT")};
    if index==0{return Ok("RECOVERABLE_NO_CURRENT")}
    if index>3{return Ok("RECOVERABLE_BAD_CURRENT")}
    let snapshot=root.join(format!("gen-{index}"));
    let manifest=snapshot.join("manifest.txt");
    if !manifest.is_file(){return Ok("RECOVERABLE_MISSING_MANIFEST")}
    let body=fs::read_to_string(manifest)?;
    let mut lines=body.lines();
    if lines.next()!=Some("group-v1") || lines.next()!=Some(index.to_string().as_str()) {return Ok("RECOVERABLE_BAD_MANIFEST")}
    for name in (1..=index).map(|n|format!("passage-{n}.mp3")).chain(["cumulative.txt".to_string(),"cumulative.srt".to_string()]) {
        let Some(line)=lines.next() else{return Ok("RECOVERABLE_BAD_MANIFEST")};
        let mut fields=line.split('\t');
        let (Some(found),Some(size),Some(expected),None)=(fields.next(),fields.next(),fields.next(),fields.next()) else{return Ok("RECOVERABLE_BAD_MANIFEST")};
        if found!=name{return Ok("RECOVERABLE_BAD_MANIFEST")}
        let path=if name.ends_with(".mp3"){root.join("assets").join(name)} else{snapshot.join(name)};
        if !path.is_file(){return Ok("RECOVERABLE_MISSING_ARTIFACT")}
        if fs::metadata(&path)?.len().to_string()!=size || hash(&path)?!=expected{return Ok("RECOVERABLE_CORRUPT_ARTIFACT")}
    }
    if lines.next().is_some(){return Ok("RECOVERABLE_BAD_MANIFEST")}
    let next=index+1;
    let pending=root.join(format!("pending-{next}.txt"));
    if pending.exists() {
        if fs::read_to_string(pending)?!=format!("pending-v1\t{next}\n") {return Ok("RECOVERABLE_BAD_PENDING")}
        return Ok("RECOVERABLE_PENDING_PASSAGE")
    }
    Ok("COMPLETE")
}
fn append(root:&Path, source:&Path, index:usize, cut:usize, raw:bool)->Result<(),Box<dyn std::error::Error>> {
    if !(1..=3).contains(&index){return Err("index 1..3".into())}
    if current(root)?==index && scan(root)?=="COMPLETE" {println!("generation={index} status=COMPLETE idempotent=true");return Ok(())}
    fs::create_dir_all(root.join("assets"))?;
    if index>1 && (current(root)?!=index-1 || !matches!(scan(root)?,"COMPLETE"|"RECOVERABLE_PENDING_PASSAGE")){return Err("prior generation invalid".into())}
    // The durable pending record precedes capture/encoding in the product contract.
    // The fixture starts from an already encoded MP3, then tests publication.
    let pending=root.join(format!("pending-{index}.txt"));
    if !pending.exists(){write_sync(&pending,format!("pending-v1\t{index}\n").as_bytes())?;}
    let name=format!("passage-{index}.mp3");
    let partial=root.join("assets").join(format!("{name}.partial"));
    if raw {encode(source,index,&partial)?;}
    else {
        let mut from=File::open(source.join(&name))?;
        let mut to=File::create(&partial)?;
        std::io::copy(&mut from,&mut to)?;
        to.sync_all()?; drop(to);
    }
    fs::rename(partial,root.join("assets").join(&name))?;
    if cut==0{std::process::exit(91)}
    let snapshot=root.join(format!("gen-{index}")); fs::create_dir_all(&snapshot)?;
    let mut txt=String::new(); let mut srt=String::new(); let mut offset=0_u64;
    for n in 1..=index {
        txt.push_str(&format!("Passage {n} fixture\n"));
        srt.push_str(&format!("{n}\n{} --> {}\nPassage {n} fixture\n\n",hms(offset),hms(offset+DURATIONS[n-1])));
        offset+=DURATIONS[n-1]+1500;
    }
    for (name,body) in [("cumulative.txt",txt),("cumulative.srt",srt)] {
        let p=snapshot.join(format!("{name}.partial")); write_sync(&p,body.as_bytes())?; fs::rename(p,snapshot.join(name))?;
    }
    if cut==1{std::process::exit(91)}
    let mut manifest=format!("group-v1\n{index}\n");
    for n in 1..=index {
        let name=format!("passage-{n}.mp3"); let p=root.join("assets").join(&name);
        manifest.push_str(&format!("{name}\t{}\t{}\n",fs::metadata(&p)?.len(),hash(&p)?));
    }
    for name in ["cumulative.txt","cumulative.srt"] {
        let p=snapshot.join(name);
        manifest.push_str(&format!("{name}\t{}\t{}\n",fs::metadata(&p)?.len(),hash(&p)?));
    }
    let p=snapshot.join("manifest.partial"); write_sync(&p,manifest.as_bytes())?; fs::rename(p,snapshot.join("manifest.txt"))?;
    if cut==2{std::process::exit(91)}
    let p=root.join("current.partial"); write_sync(&p,format!("{index}\n").as_bytes())?;
    if cut==3{std::process::exit(91)}
    fs::rename(p,root.join("current.txt"))?;
    if cut==4{std::process::exit(91)}
    assert_eq!(scan(root)?,"COMPLETE");
    println!("generation={index} status=COMPLETE start_ms={} end_ms={}",offset-DURATIONS[index-1]-1500,offset-1500);
    Ok(())
}
fn main()->Result<(),Box<dyn std::error::Error>> {
    let args:Vec<String>=env::args().collect();
    match args.as_slice(){
        [_,mode,root,source,index,cut] if mode=="append"=>append(Path::new(root),Path::new(source),index.parse()?,cut.parse()?,false),
        [_,mode,root,source,index,cut] if mode=="append_raw"=>append(Path::new(root),Path::new(source),index.parse()?,cut.parse()?,true),
        [_,mode,root] if mode=="scan"=>{println!("generation={} status={}",current(Path::new(root)).unwrap_or(0),scan(Path::new(root))?);Ok(())},
        _=>Err("usage: e3-resume append ROOT SOURCE INDEX CUT | scan ROOT".into())
    }
}
