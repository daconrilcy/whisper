use std::{fs::File, path::Path};
use symphonia::core::{errors::Error, formats::FormatOptions, io::MediaSourceStream, meta::MetadataOptions, probe::Hint, codecs::DecoderOptions};
fn main() -> Result<(), Box<dyn std::error::Error>> {
 for arg in std::env::args().skip(1) {
  let path=Path::new(&arg);
  let source=MediaSourceStream::new(Box::new(File::open(path)?),Default::default());
  let mut hint=Hint::new();
  if let Some(ext)=path.extension().and_then(|x| x.to_str()) { hint.with_extension(ext); }
  let probe=symphonia::default::get_probe().format(&hint,source,&FormatOptions::default(),&MetadataOptions::default())?;
  let mut format=probe.format;
  let track=format.default_track().ok_or("no track")?;
  let id=track.id;
  let rate=track.codec_params.sample_rate.unwrap_or(0);
  let channels=track.codec_params.channels.map(|x|x.count()).unwrap_or(0);
  let mut decoder=symphonia::default::get_codecs().make(&track.codec_params,&DecoderOptions::default())?;
  let mut frames=0u64;
  let mut packets=0u64;
  loop { match format.next_packet() {
   Ok(packet) if packet.track_id()==id => { let decoded=decoder.decode(&packet)?; frames+=decoded.frames() as u64; packets+=1; },
   Ok(_) => {},
   Err(Error::IoError(e)) if e.kind()==std::io::ErrorKind::UnexpectedEof => break,
   Err(e) => return Err(e.into()),
  }}
  println!("{} rate={} channels={} frames={} packets={} duration_ms={}",path.display(),rate,channels,frames,packets,frames*1000/u64::from(rate));
 }
 Ok(())
}
