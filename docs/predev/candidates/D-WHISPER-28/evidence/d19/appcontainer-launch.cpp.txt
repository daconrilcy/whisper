#define UNICODE
#define _UNICODE
#include <windows.h>
#include <userenv.h>
#include <sddl.h>
#include <iostream>
#include <string>
#include <vector>
#pragma comment(lib, "userenv.lib")
#pragma comment(lib, "advapi32.lib")

static std::wstring quote(const std::wstring& value) { return L"\"" + value + L"\""; }
static void fail(const wchar_t* stage, DWORD code) { std::wcerr << stage << L" error=" << code << L"\n"; }

int wmain(int argc, wchar_t** argv) {
    if (argc < 3) { std::wcerr << L"usage: appcontainer_launch create|run|delete PROFILE [EXE MODEL WAV MODE]\n"; return 2; }
    std::wstring action = argv[1], profile = argv[2];
    if (action == L"delete") {
        HRESULT hr = DeleteAppContainerProfile(profile.c_str());
        std::wcout << L"delete_hr=" << std::hex << hr << L"\n";
        return FAILED(hr) ? 1 : 0;
    }
    PSID sid = nullptr;
    if (action == L"create") {
        HRESULT hr = CreateAppContainerProfile(profile.c_str(), profile.c_str(), L"Whisper offline design fixture",
                                               nullptr, 0, &sid);
        if (FAILED(hr)) { std::wcerr << L"create_hr=" << std::hex << hr << L"\n"; return 1; }
    } else if (action == L"run") {
        HRESULT hr = DeriveAppContainerSidFromAppContainerName(profile.c_str(), &sid);
        if (FAILED(hr)) { std::wcerr << L"derive_hr=" << std::hex << hr << L"\n"; return 1; }
    } else { return 2; }
    LPWSTR sid_text = nullptr;
    if (!ConvertSidToStringSidW(sid, &sid_text)) { fail(L"sid_string", GetLastError()); FreeSid(sid); return 1; }
    std::wcout << L"appcontainer_sid=" << sid_text << L" capabilities=0\n";
    LocalFree(sid_text);
    if (action == L"create") { FreeSid(sid); return 0; }
    if (argc != 7) { FreeSid(sid); return 2; }
    SIZE_T length = 0;
    InitializeProcThreadAttributeList(nullptr, 1, 0, &length);
    std::vector<unsigned char> storage(length);
    auto attributes = reinterpret_cast<LPPROC_THREAD_ATTRIBUTE_LIST>(storage.data());
    if (!InitializeProcThreadAttributeList(attributes, 1, 0, &length)) { fail(L"init_attributes", GetLastError()); FreeSid(sid); return 1; }
    SECURITY_CAPABILITIES capabilities{};
    capabilities.AppContainerSid = sid;
    capabilities.CapabilityCount = 0;
    if (!UpdateProcThreadAttribute(attributes, 0, PROC_THREAD_ATTRIBUTE_SECURITY_CAPABILITIES,
                                   &capabilities, sizeof(capabilities), nullptr, nullptr)) {
        fail(L"update_attributes", GetLastError()); DeleteProcThreadAttributeList(attributes); FreeSid(sid); return 1;
    }
    STARTUPINFOEXW startup{};
    startup.StartupInfo.cb = sizeof(startup);
    startup.StartupInfo.dwFlags = STARTF_USESTDHANDLES;
    startup.StartupInfo.hStdInput = GetStdHandle(STD_INPUT_HANDLE);
    startup.StartupInfo.hStdOutput = GetStdHandle(STD_OUTPUT_HANDLE);
    startup.StartupInfo.hStdError = GetStdHandle(STD_ERROR_HANDLE);
    startup.lpAttributeList = attributes;
    PROCESS_INFORMATION process{};
    std::wstring executable = argv[3];
    std::wstring command = quote(executable) + L" " + quote(argv[4]) + L" " + quote(argv[5]) + L" " + argv[6];
    std::vector<wchar_t> command_mut(command.begin(), command.end());
    command_mut.push_back(L'\0');
    BOOL launched = CreateProcessW(executable.c_str(), command_mut.data(), nullptr, nullptr, TRUE,
        EXTENDED_STARTUPINFO_PRESENT, nullptr, nullptr, &startup.StartupInfo, &process);
    if (!launched) { fail(L"launch", GetLastError()); DeleteProcThreadAttributeList(attributes); FreeSid(sid); return 1; }
    HANDLE token = nullptr;
    DWORD is_container = 0, size = 0;
    if (!OpenProcessToken(process.hProcess, TOKEN_QUERY, &token) ||
        !GetTokenInformation(token, TokenIsAppContainer, &is_container, sizeof(is_container), &size)) {
        fail(L"token", GetLastError());
    } else { std::wcout << L"child_pid=" << process.dwProcessId << L" token_is_appcontainer=" << is_container << L"\n"; }
    if (token) CloseHandle(token);
    WaitForSingleObject(process.hProcess, INFINITE);
    DWORD code = 0;
    GetExitCodeProcess(process.hProcess, &code);
    std::wcout << L"child_exit=" << code << L"\n";
    CloseHandle(process.hThread); CloseHandle(process.hProcess);
    DeleteProcThreadAttributeList(attributes); FreeSid(sid);
    return code == 0 ? 0 : 1;
}
