function Assert-FrozenPack {
    $manifestPath = Join-Path $pack 'pack-manifest.json'
    if (-not (Test-Path -LiteralPath $manifestPath -PathType Leaf)) { throw 'Frozen pack manifest is missing.' }
    if ((Get-FileHash -Algorithm SHA256 -LiteralPath $manifestPath).Hash.ToLowerInvariant() -ne $expectedPackManifestSha) {
        throw 'Frozen pack manifest differs from the qualified SHA-256.'
    }
    $manifest = Get-Content -LiteralPath $manifestPath -Raw | ConvertFrom-Json
    if ($manifest.digest -ne $expectedPackDigest -or $manifest.files.Count -ne 24) {
        throw 'Frozen pack identity differs from the qualified manifest.'
    }
    $expectedPaths = @{}
    foreach ($entry in $manifest.files) {
        $relative = [string]$entry.path
        if ($relative.StartsWith('/') -or $relative.Contains('\') -or $relative.Split('/') -contains '..' -or $expectedPaths.ContainsKey($relative)) {
            throw 'Frozen pack contains an unsafe or duplicate path.'
        }
        $expectedPaths[$relative] = $true
        $file = Join-Path $pack $relative.Replace('/', '\')
        if (-not (Test-Path -LiteralPath $file -PathType Leaf)) { throw "Frozen pack file missing: $relative" }
        if ((Get-FileHash -Algorithm SHA256 -LiteralPath $file).Hash.ToLowerInvariant() -ne $entry.sha256) {
            throw "Frozen pack file changed: $relative"
        }
    }
    $actualPaths = @{}
    foreach ($item in Get-ChildItem -LiteralPath $pack -Force -Recurse) {
        if ($item.Attributes -band [System.IO.FileAttributes]::ReparsePoint) { throw 'Frozen pack contains a link or junction.' }
        if (-not $item.PSIsContainer) {
            $relative = [System.IO.Path]::GetRelativePath($pack, $item.FullName).Replace('\', '/')
            $actualPaths[$relative] = $true
        }
    }
    if ($actualPaths.Count -ne $expectedPaths.Count + 1 -or -not $actualPaths.ContainsKey('pack-manifest.json')) {
        throw 'Frozen pack contains an unexpected or missing file.'
    }
    foreach ($relative in $expectedPaths.Keys) {
        if (-not $actualPaths.ContainsKey($relative)) { throw "Frozen pack inventory mismatch: $relative" }
    }
}
